cmake_minimum_required(VERSION 3.16)
project(codec_failure_test C)
enable_testing()
add_executable(codec_failure_test test_codec.c ../../esp_codec_dev.c)
target_include_directories(codec_failure_test PRIVATE include ../.. ../../include ../../interface)
if(MSVC)
    target_compile_options(codec_failure_test PRIVATE /UNDEBUG)
else()
    target_compile_options(codec_failure_test PRIVATE -UNDEBUG)
endif()
add_test(NAME codec_failure_test COMMAND codec_failure_test)
set(I2S_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../../platform/audio_codec_data_i2s.c")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${I2S_SOURCE}")
file(READ "${I2S_SOURCE}" I2S_BODY)
string(FIND "${I2S_BODY}" "static int _i2s_data_enable(" FIRST)
string(FIND "${I2S_BODY}" "static int _i2s_data_read(" LAST)
math(EXPR LENGTH "${LAST}-${FIRST}")
string(SUBSTRING "${I2S_BODY}" ${FIRST} ${LENGTH} PART)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/i2s_privacy.inc" "${PART}")
add_executable(i2s_privacy_test test_i2s_privacy.c)
target_include_directories(i2s_privacy_test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME i2s_privacy_test COMMAND i2s_privacy_test)
add_executable(i2s_privacy_hw2_test test_i2s_privacy.c)
target_compile_definitions(i2s_privacy_hw2_test PRIVATE SOC_I2S_HW_VERSION_1=0 SOC_I2S_HW_VERSION_2=1)
target_include_directories(i2s_privacy_hw2_test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME i2s_privacy_hw2_test COMMAND i2s_privacy_hw2_test)
