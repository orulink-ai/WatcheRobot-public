cmake_minimum_required(VERSION 3.20)
project(mcu_uart_gate_tests C CXX)
enable_testing()
find_package(Threads REQUIRED)
add_executable(mcu_uart_gate_tests ../../src/mcu_link_uart.c test_uart_gate.cpp)
target_include_directories(mcu_uart_gate_tests PRIVATE include ../../include)
target_compile_features(mcu_uart_gate_tests PRIVATE c_std_11 cxx_std_17)
if(MSVC)
    target_compile_options(mcu_uart_gate_tests PRIVATE /UNDEBUG)
else()
    target_compile_options(mcu_uart_gate_tests PRIVATE -UNDEBUG)
endif()
target_link_libraries(mcu_uart_gate_tests PRIVATE Threads::Threads)
add_test(NAME mcu_uart_gate_tests COMMAND mcu_uart_gate_tests)
set_tests_properties(mcu_uart_gate_tests PROPERTIES TIMEOUT 10)
