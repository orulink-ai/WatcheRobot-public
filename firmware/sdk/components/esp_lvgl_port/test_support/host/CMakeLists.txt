cmake_minimum_required(VERSION 3.20)
project(lcd_dma_host_tests C)
enable_testing()
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
# Test actual production functions, not separately maintained copies.
set(source_path "${CMAKE_CURRENT_SOURCE_DIR}/../../esp_lvgl_port.c")
file(READ "${source_path}" source)
string(REPLACE "\r\n" "\n" source "${source}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${source_path}")
foreach(name remove_disp flush_staged)
    if(name STREQUAL "remove_disp")
        set(signature "esp_err_t lvgl_port_remove_disp(")
        set(output_name lcd_remove_disp)
    else()
        set(signature "static esp_err_t lvgl_port_flush_staged(")
        set(output_name lcd_flush_staged)
    endif()
    string(FIND "${source}" "${signature}" begin)
    if(begin LESS 0)
        message(FATAL_ERROR "Missing production function ${signature}")
    endif()
    string(SUBSTRING "${source}" ${begin} -1 tail)
    string(FIND "${tail}" "\n}" end)
    if(end LESS 0)
        message(FATAL_ERROR "Unterminated production function ${signature}")
    endif()
    math(EXPR length "${end} + 3")
    string(SUBSTRING "${tail}" 0 ${length} body)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${output_name}.inc" "${body}\n")
endforeach()
foreach(name release staging)
    add_executable(lcd_dma_${name} test_lcd_dma_${name}.c)
    target_include_directories(lcd_dma_${name} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
    if(MSVC)
        target_compile_options(lcd_dma_${name} PRIVATE /UNDEBUG)
    else()
        target_compile_options(lcd_dma_${name} PRIVATE -UNDEBUG)
    endif()
    add_test(NAME lcd_dma_${name} COMMAND lcd_dma_${name})
endforeach()
