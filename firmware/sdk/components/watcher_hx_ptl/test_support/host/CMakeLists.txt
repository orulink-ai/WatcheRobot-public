cmake_minimum_required(VERSION 3.20)
project(hx_control_host_tests C)
enable_testing()
add_executable(hx_control_tests test_control.c ../../watcher_hx_control.c ../../../hx_ptl_protocol/hx_vision_control_codec.c)
target_include_directories(hx_control_tests PRIVATE stubs ../../include ../../../hx_ptl_protocol/include)
add_test(NAME hx_control_tests COMMAND hx_control_tests)

# Exercise the production PTL cold-start sequence without a second board owner.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../watcher_hx_ptl.c" ptl_source)
string(REPLACE "\r\n" "\n" ptl_source "${ptl_source}")
string(FIND "${ptl_source}" "esp_err_t watcher_hx_ptl_init(void) {" begin)
if(begin LESS 0)
    message(FATAL_ERROR "PTL initialization function missing")
endif()
string(SUBSTRING "${ptl_source}" ${begin} -1 tail)
string(FIND "${tail}" "\n}" end)
math(EXPR length "${end} + 3")
string(SUBSTRING "${tail}" 0 ${length} ptl_init)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/ptl_init.inc" "${ptl_init}\n")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/../../watcher_hx_ptl.c")
add_executable(hx_ptl_init_tests test_ptl_init.c)
target_include_directories(hx_ptl_init_tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
if(MSVC)
    target_compile_options(hx_ptl_init_tests PRIVATE /W4 /WX)
else()
    target_compile_options(hx_ptl_init_tests PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
add_test(NAME hx_ptl_init_tests COMMAND hx_ptl_init_tests)
