cmake_minimum_required(VERSION 3.20)
project(hardware_demo_state_tests C)
enable_testing()
add_executable(demo_state_tests test_demo_state.c ../../demo_state.c)
target_include_directories(demo_state_tests PRIVATE ../..)
set_target_properties(demo_state_tests PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES)
if(MSVC)
  target_compile_options(demo_state_tests PRIVATE /W4)
else()
  target_compile_options(demo_state_tests PRIVATE -Wall -Wextra -Werror)
endif()
