cmake_minimum_required(VERSION 3.20)
project(check_view_real_lvgl_tests LANGUAGES C CXX)

enable_testing()
set(LVGL_SOURCE_DIR "" CACHE PATH "Actual LVGL 8.4.0 source directory (not an ESP component shim)")
option(HOST_WRITE_PPM "Write software snapshots in the build directory" OFF)
if(NOT EXISTS "${LVGL_SOURCE_DIR}/lvgl.h" OR NOT EXISTS "${LVGL_SOURCE_DIR}/src/core/lv_obj.c")
  message(FATAL_ERROR "Pass -DLVGL_SOURCE_DIR=<actual managed_components/lvgl__lvgl directory>")
endif()
file(STRINGS "${LVGL_SOURCE_DIR}/lvgl.h" lvgl_version REGEX "^#define LVGL_VERSION_(MAJOR|MINOR|PATCH) ")
message(STATUS "Actual LVGL: ${lvgl_version}")
if(NOT lvgl_version MATCHES "LVGL_VERSION_MAJOR 8([ ;]|$)" OR
   NOT lvgl_version MATCHES "LVGL_VERSION_MINOR 4([ ;]|$)" OR
   NOT lvgl_version MATCHES "LVGL_VERSION_PATCH 0([ ;]|$)")
  message(FATAL_ERROR "Firmware contract requires actual LVGL 8.4.0; supplied library version does not match")
endif()

set(LV_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE STRING "" FORCE)
set(LV_CONF_INCLUDE_SIMPLE ON CACHE BOOL "" FORCE)
add_subdirectory("${LVGL_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/lvgl" EXCLUDE_FROM_ALL)
get_filename_component(view_dir "${CMAKE_CURRENT_SOURCE_DIR}/../../main" ABSOLUTE)
get_filename_component(common_dir "${CMAKE_CURRENT_SOURCE_DIR}/../../../common" ABSOLUTE)

# Production sources remain unchanged; warnings-as-errors apply to our harness
# and mocks, not the existing firmware or upstream LVGL sources.
add_library(check_view_under_test STATIC "${view_dir}/check_view.c"
  "${view_dir}/check_layout.c" "${common_dir}/check_flow.c" "${common_dir}/demo_state.c")
target_link_libraries(check_view_under_test PUBLIC lvgl::lvgl)
add_executable(check_view_tests test_check_view.c host_mocks.c)
target_link_libraries(check_view_tests PRIVATE check_view_under_test)
foreach(target check_view_tests check_view_under_test)
  target_include_directories(${target} PRIVATE include "${view_dir}" "${common_dir}")
  set_target_properties(${target} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
endforeach()
if(MSVC)
  target_compile_options(check_view_tests PRIVATE /W4 /WX /UNDEBUG)
  target_compile_options(check_view_under_test PRIVATE /W4 /UNDEBUG)
  target_compile_options(lvgl PRIVATE /UNDEBUG)
else()
  target_compile_options(check_view_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  target_compile_options(check_view_under_test PRIVATE -Wall -Wextra -UNDEBUG)
  target_compile_options(lvgl PRIVATE -UNDEBUG)
  target_link_libraries(check_view_under_test PUBLIC m)
endif()
if(HOST_WRITE_PPM)
  file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/snapshots")
endif()
foreach(scenario config geometry_oracle nav_geometry nav_pixels labels content status motion_safety running_content report detail_tail stop stopping pressed busy native_input release_outside release_disabled photo live_image stale_image image_arrival_attempt)
  if(HOST_WRITE_PPM)
    add_test(NAME check_view_${scenario} COMMAND check_view_tests ${scenario}
      --ppm "${CMAKE_CURRENT_BINARY_DIR}/snapshots/${scenario}.ppm")
  else()
    add_test(NAME check_view_${scenario} COMMAND check_view_tests ${scenario})
  endif()
  set_tests_properties(check_view_${scenario} PROPERTIES TIMEOUT 30)
endforeach()
